Cómo reproducir acciones al azar en Godot 4 (GDScript)

Guarda el archivo exportado como res://assets/audio/audio_actions.json y usa este helper para reproducir un sonido aleatorio de cada acción: 

var _audio_actions: Dictionary = {}
var _stream_cache: Dictionary = {}

func _ready() -> void:
	var file := FileAccess.open("res://assets/audio/audio_actions.json", FileAccess.READ)
	if file:
		var parsed: Variant = JSON.parse_string(file.get_as_text())
		if parsed is Dictionary and parsed.has("actions"):
			_audio_actions = parsed["actions"]

func play_random_action_sound(action_id: String, player: AudioStreamPlayer3D) -> void:
	if not _audio_actions.has(action_id):
		return
	var sounds: Array = _audio_actions[action_id].get("sounds", [])
	if sounds.is_empty():
		return
	var chosen_path: String = sounds.pick_random()
	if not _stream_cache.has(chosen_path):
		_stream_cache[chosen_path] = load(chosen_path)
	player.stream = _stream_cache[chosen_path]
	player.play()
